client/src/pages/AdminJobApplications.tsx

CURRENT STATE

The Resume / CV is correctly displayed inside Application Details:

Resume / CV
legaspi_resume.pdf
Submitted with application
View

However clicking View or Download fails.

Browser console shows:

Failed to load resource: the server responded with a status of 401 (Unauthorized)

The resume endpoint appears to be something similar to:

/api/admin/job-applications/:applicationId/resume

ROOT CAUSE TO VERIFY

The admin application APIs authenticate using the JWT stored in:

localStorage.getItem("onspot_jwt_token")

Normal API requests in AdminJobApplications.tsx already send:

Authorization: Bearer <token>

But opening a protected resume URL directly with:

window.open(url)

or:

<a href={url}>

does NOT include the Authorization header.

Therefore the protected resume endpoint returns 401.

Do not make the resume endpoint public just to fix this.

==================================================
REQUIRED FIX
==================================================

Implement authenticated Resume View and Download using fetch().

Create a helper in AdminJobApplications.tsx such as:

async function fetchResumeBlob(applicationId: string) {
  const token = localStorage.getItem("onspot_jwt_token");

  const res = await fetch(
    `/api/admin/job-applications/${applicationId}/resume`,
    {
      headers: {
        ...(token
          ? { Authorization: `Bearer ${token}` }
          : {}),
      },
    }
  );

  if (!res.ok) {
    const message =
      res.status === 401
        ? "You are not authorized to view this resume."
        : "Unable to load resume.";

    throw new Error(message);
  }

  return await res.blob();
}

Do NOT use apiFetch() if apiFetch always assumes JSON responses.

The resume endpoint returns binary data, not JSON.

==================================================
VIEW CV
==================================================

When clicking:

View

do NOT do:

window.open(resumeUrl)

if resumeUrl is a protected API endpoint.

Instead:

1. Fetch the resume using authenticated fetch.
2. Convert the response to a Blob.
3. Create an object URL.
4. Open that object URL in a new browser tab.

Example:

const handleViewResume = async () => {
  try {
    setResumeAction("view");

    const token = localStorage.getItem("onspot_jwt_token");

    const res = await fetch(
      `/api/admin/job-applications/${detail.id}/resume`,
      {
        headers: {
          ...(token
            ? { Authorization: `Bearer ${token}` }
            : {}),
        },
      }
    );

    if (!res.ok) {
      throw new Error(
        res.status === 401
          ? "Unauthorized to view resume"
          : "Unable to load resume"
      );
    }

    const blob = await res.blob();

    const objectUrl = URL.createObjectURL(blob);

    const newWindow = window.open(
      objectUrl,
      "_blank",
      "noopener,noreferrer"
    );

    if (!newWindow) {
      URL.revokeObjectURL(objectUrl);
      throw new Error(
        "Your browser blocked the resume preview. Please allow popups or use Download."
      );
    }

    // Do not revoke immediately because the new tab needs it.
    setTimeout(() => {
      URL.revokeObjectURL(objectUrl);
    }, 60_000);

  } catch (error: any) {
    toast({
      title: "Unable to open resume",
      description: error.message,
      variant: "destructive",
    });
  } finally {
    setResumeAction(null);
  }
};

DOWNLOAD CV
==================================================

The Download button must also use authenticated fetch.

Do not use:

<a href="/api/admin/.../resume" download>

because it will also lose the Authorization header.

Instead:

const handleDownloadResume = async () => {
  try {
    setResumeAction("download");

    const token = localStorage.getItem("onspot_jwt_token");

    const res = await fetch(
      `/api/admin/job-applications/${detail.id}/resume`,
      {
        headers: {
          ...(token
            ? { Authorization: `Bearer ${token}` }
            : {}),
        },
      }
    );

    if (!res.ok) {
      throw new Error(
        res.status === 401
          ? "Unauthorized to download resume"
          : "Unable to download resume"
      );
    }

    const blob = await res.blob();
    const objectUrl = URL.createObjectURL(blob);

    const anchor = document.createElement("a");
    anchor.href = objectUrl;
    anchor.download =
      detail.resumeFileName || "resume";
    document.body.appendChild(anchor);
    anchor.click();
    anchor.remove();

    URL.revokeObjectURL(objectUrl);

  } catch (error: any) {
    toast({
      title: "Unable to download resume",
      description: error.message,
      variant: "destructive",
    });
  } finally {
    setResumeAction(null);
  }
};

==================================================
BETTER IMPLEMENTATION — AVOID DUPLICATION
==================================================

Prefer one reusable function:

async function getResumeBlob(applicationId: string) {
  const token = localStorage.getItem("onspot_jwt_token");

  const response = await fetch(
    `/api/admin/job-applications/${applicationId}/resume`,
    {
      method: "GET",
      headers: {
        ...(token
          ? { Authorization: `Bearer ${token}` }
          : {}),
      },
    }
  );

  if (!response.ok) {
    const body = await response.json().catch(() => null);

    throw new Error(
      body?.message ||
      body?.error ||
      `Failed to load resume (${response.status})`
    );
  }

  return response.blob();
}

Then both View and Download use that helper.
