<div className="absolute inset-x-0 top-0 h-px bg-gradient-to-r from-transparent via-[#474ead]/25 to-transparent" />

<div className="inline-flex items-center rounded-full border border-[#474ead]/20 bg-[#474ead]/5 px-4 py-2 text-sm font-medium text-[#474ead] shadow-sm">
