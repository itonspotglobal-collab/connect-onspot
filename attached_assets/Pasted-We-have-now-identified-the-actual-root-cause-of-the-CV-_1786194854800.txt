==================================================
VIEW CV
==================================================

Once JWT auth is fixed:

View should still fetch the protected file first.

Use:

const previewWindow = window.open("", "_blank");

try {
  const token =
    localStorage.getItem("onspot_jwt_token");

  if (!token) {
    previewWindow?.close();
    throw new Error(
      "Your admin session has expired. Please sign in again."
    );
  }

  const response = await fetch(
    `/api/admin/job-applications/${detail.id}/resume`,
    {
      headers: {
        Authorization: `Bearer ${token}`,
      },
    }
  );

  if (response.status === 401 ||
      response.status === 403) {

    previewWindow?.close();

    throw new Error(
      "Your admin session has expired or you do not have permission."
    );
  }

  if (!response.ok) {
    throw new Error("Unable to load resume.");
  }

  const blob = await response.blob();

  const objectUrl =
    URL.createObjectURL(blob);

  if (previewWindow) {
    previewWindow.location.href = objectUrl;
  }

  setTimeout(
    () => URL.revokeObjectURL(objectUrl),
    60000
  );

} catch (error) {
  previewWindow?.close();
  // toast
}
