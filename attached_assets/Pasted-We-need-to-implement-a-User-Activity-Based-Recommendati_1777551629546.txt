Suggested helper implementation:

const STORAGE_KEY = "userActivityMemory";
const MAX_ACTIVITIES = 50;

export function saveUserActivity(activity) {
  try {
    if (typeof window === "undefined") return;

    const activities = getUserActivities();

    const newActivity = {
      ...activity,
      createdAt: activity.createdAt || new Date().toISOString(),
    };

    const isDuplicate = activities.some(existing => {
      const sameType = existing.activityType === newActivity.activityType;
      const sameReference = existing.referenceId && newActivity.referenceId && existing.referenceId === newActivity.referenceId;
      const sameKeyword = existing.keyword && newActivity.keyword && existing.keyword.toLowerCase() === newActivity.keyword.toLowerCase();

      const existingTime = new Date(existing.createdAt).getTime();
      const newTime = new Date(newActivity.createdAt).getTime();
      const withinOneMinute = Math.abs(newTime - existingTime) < 60 * 1000;

      return sameType && withinOneMinute && (sameReference || sameKeyword);
    });

    if (isDuplicate) return;

    const updated = [newActivity, ...activities].slice(0, MAX_ACTIVITIES);
    localStorage.setItem(STORAGE_KEY, JSON.stringify(updated));
  } catch (error) {
    console.warn("Unable to save user activity", error);
  }
}

export function getUserActivities() {
  try {
    if (typeof window === "undefined") return [];

    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return [];

    const parsed = JSON.parse(raw);
    return Array.isArray(parsed) ? parsed : [];
  } catch (error) {
    console.warn("Unable to read user activities", error);
    return [];
  }
}

export function clearUserActivities() {
  try {
    if (typeof window === "undefined") return;
    localStorage.removeItem(STORAGE_KEY);
  } catch (error) {
    console.warn("Unable to clear user activities", error);
  }
}

function normalizeTerm(term) {
  if (!term || typeof term !== "string") return "";
  return term.trim().toLowerCase();
}

function getActivityWeight(activityType) {
  const weights = {
    JobSearch: 5,
    TalentSearch: 5,
    JobClick: 4,
    JobView: 4,
    TalentView: 4,
    ArticleView: 2,
    CategoryClick: 3,
    FilterClick: 3,
    SavedJob: 7,
    AppliedJob: 8,
  };

  return weights[activityType] || 1;
}

function getRecencyMultiplier(createdAt) {
  const created = new Date(createdAt).getTime();
  if (!created) return 0.5;

  const now = Date.now();
  const ageInMs = now - created;
  const oneDay = 24 * 60 * 60 * 1000;
  const sevenDays = 7 * oneDay;

  if (ageInMs <= oneDay) return 1;
  if (ageInMs <= sevenDays) return 0.75;
  return 0.5;
}

function extractTerms(activity) {
  const terms = [];

  if (activity.keyword) terms.push(activity.keyword);
  if (activity.title) terms.push(activity.title);
  if (activity.category) terms.push(activity.category);

  if (Array.isArray(activity.tags)) {
    terms.push(...activity.tags);
  }

  if (Array.isArray(activity.skills)) {
    terms.push(...activity.skills);
  }

  return terms
    .map(normalizeTerm)
    .filter(Boolean);
}

export function getUserInterestProfile() {
  const activities = getUserActivities();
  const profile = {};

  activities.forEach(activity => {
    const weight = getActivityWeight(activity.activityType);
    const recencyMultiplier = getRecencyMultiplier(activity.createdAt);
    const score = weight * recencyMultiplier;
    const terms = extractTerms(activity);

    terms.forEach(term => {
      profile[term] = (profile[term] || 0) + score;
    });
  });

  return Object.entries(profile)
    .sort((a, b) => b[1] - a[1])
    .map(([term, score]) => ({
      term,
      score,
    }));
}

export function getTopUserInterests(limit = 5) {
  return getUserInterestProfile()
    .slice(0, limit)
    .map(item => item.term);
}

export function getRecommendedSearchTerms(limit = 3) {
  return getTopUserInterests(limit).join(" ");
}
