const currencySymbols: Record<string, string> = {
  PHP: "₱",
  USD: "$",
  EUR: "€",
  GBP: "£",
  AUD: "A$",

function getCurrencyCode(job: JobPosting) {
  return job.currency === "OTHER"
    ? job.customCurrencyCode || ""
    : job.currency || "PHP";
}

function formatJobCurrency(
  amount: number,
  currency: SupportedCurrency,
  customCurrencyCode?: string
) {
  const code =
    currency === "OTHER"
      ? customCurrencyCode || "PHP"
      : currency || "PHP";

  try {
    return new Intl.NumberFormat("en-US", {
      style: "currency",
      currency: code,
      maximumFractionDigits: 0,
    }).format(amount);
  } catch {
    return `${code} ${amount.toLocaleString()}`;
  }
}

Centralize salary display logic
Create a helper such as:
function formatJobSalary(job: JobPosting): string {
  const code =
    job.currency === "OTHER"
      ? job.customCurrencyCode || "PHP"
      : job.currency || "PHP";

  if (job.fixedMonthlyBudget) {
    return `${formatJobCurrency(
      job.fixedMonthlyBudget,
      job.currency,
      job.customCurrencyCode
    )}/mo`;
  }

  if (job.rateRangeMin && job.rateRangeMax) {
    return `${formatJobCurrency(
      job.rateRangeMin,
      job.currency,
      job.customCurrencyCode
    )}–${formatJobCurrency(
      job.rateRangeMax,
      job.currency,
      job.customCurrencyCode
    )}/mo`;
  }

  if (job.rateRangeMin) {
    return `${formatJobCurrency(
      job.rateRangeMin,
      job.currency,
      job.customCurrencyCode
    )}+/mo`;
  }

  return "Rate TBD";
}
