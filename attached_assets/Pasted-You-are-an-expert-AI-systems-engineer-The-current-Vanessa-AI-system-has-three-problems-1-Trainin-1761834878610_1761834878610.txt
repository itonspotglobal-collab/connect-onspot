function extractTopicFromText(text) {
  const words = ["CEO", "Delivery", "Operations", "Pricing", "Office", "Hiring", "OnSpot"];
  const found = words.find((w) => text.toLowerCase().includes(w.toLowerCase()));
  return found || "general";
}

router.delete("/api/vanessa/responses/:threadId", async (req, res) => {
  const { threadId } = req.params;
  try {
    await storage.deleteThread(threadId);
    res.json({ success: true, message: `Deleted thread ${threadId}` });
  } catch (error) {
    console.error("❌ Error deleting thread:", error);
    res.status(500).json({ success: false, message: "Failed to delete thread" });
  }
});
