```ts
  useState<string | null>(null);
```

```ts

```


# Important Timer Cleanup

Avoid stacking timers if the user clicks several navigation items quickly.

Use a ref:

```ts
const highlightTimerRef =
  useRef<ReturnType<typeof setTimeout> | null>(null);
```

Before starting another timer:

```ts
if (highlightTimerRef.current) {
  clearTimeout(highlightTimerRef.current);
}
```

Then create a new timer.

Also clear it on unmount.

# Recommended Implementation Example

Conceptually:

```tsx
const [highlightedSectionId, setHighlightedSectionId] =
  useState<string | null>(null);

const highlightTimerRef = useRef<number | null>(null);

function highlightSection(id: string) {
  setHighlightedSectionId(id);

  if (highlightTimerRef.current) {
    window.clearTimeout(highlightTimerRef.current);
  }

  highlightTimerRef.current = window.setTimeout(() => {
    setHighlightedSectionId(null);
  }, 1400);
}
```

Then:

```tsx
<Section
  id="section-skills"
  title="Skills"
  icon={Star}
  highlighted={highlightedSectionId === "section-skills"}
>
```

Do the same for each reusable `Section`.
